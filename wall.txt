def find_daemon():
    """linphoned.exe / linphone-daemon.exe dhoondo. None = nahi mila."""
    env = os.environ.get("LINPHONED_PATH", "")
    cands = [Path(env)] if env else []
    for base in (r"C:\Program Files\Linphone\bin", r"C:\Program Files (x86)\Linphone\bin",
                 r"C:\Program Files\Linphone", r"C:\Program Files (x86)\Linphone"):
        for name in ("linphoned.exe", "linphone-daemon.exe"):
            cands.append(Path(base) / name)
    for c in cands:
        if c and c.exists():
            return c
    return None
def find_sh(daemon):

def _tokens(text):
    # Sirf akshar/ank/nishan (matra) rakho — Devanagari ki matrayein \w me
    # nahi aati, isliye category se chhanta hai, regex se nahi.
    out = []
    for ch in text.lower():
        if ch.isspace() or unicodedata.category(ch)[0] in "LNM":
            out.append(ch)
        else:
            out.append(" ")
    return "".join(out).split()
def _find(tokens, words):
    """words wale kisi token ka pehla index, warna -1."""
    for i, tok in enumerate(tokens):
        if tok in words:
            return i
    return -1
def _target(tokens):
    """(kind, value, display) ya None. 2-shabd naam pehle check hota hai."""
    for n in (2, 1):
        for i in range(len(tokens) - n + 1):
            name = " ".join(tokens[i:i + n])
            if name in TARGETS:
                return TARGETS[name]
    return None
def _neg_before(tokens, upto):
    return any(tok in NEG for tok in tokens[:upto])
