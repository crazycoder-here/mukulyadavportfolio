import { db } from "@/db";
import { shouts } from "@/db/schema";
import { desc } from "drizzle-orm";
export const dynamic = "force-dynamic";
export async function GET() {
  try {
    const list = await db
      .select()
      .from(shouts)
      .orderBy(desc(shouts.createdAt))
      .limit(30);
    return Response.json(list);
  } catch (error) {
    console.error("Failed to fetch shouts:", error);
    return Response.json({ error: "Failed to fetch shouts" }, { status: 500 });
  }
}
export async function POST(request: Request) {
  try {
    const body = await request.json();
    const { name, message, songTitle, songArtist } = body;
    if (!message || message.trim() === "") {
      return Response.json({ error: "Message is required" }, { status: 400 });
    }
    const cleanName = name && name.trim() !== "" ? name.trim().slice(0, 50) : "Anonymous";
    const cleanMessage = message.trim().slice(0, 300);
    const [newShout] = await db
      .insert(shouts)
      .values({
        name: cleanName,
        message: cleanMessage,
        songTitle: songTitle || null,
        songArtist: songArtist || null,
      })
      .returning();
    return Response.json(newShout);
  } catch (error) {
    console.error("Failed to post shout:", error);
    return Response.json({ error: "Failed to save memory" }, { status: 500 });
  }
}
